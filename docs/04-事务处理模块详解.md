# TiKV 事务处理模块详解

## 1. 模块概述

TiKV 的事务处理模块是整个系统的核心组件之一，实现了分布式事务的完整生命周期管理。该模块基于 MVCC (Multi-Version Concurrency Control) 模型，支持 ACID 特性，提供了高效的事务处理能力。

### 1.1 设计目标

- **ACID 保证**: 提供原子性、一致性、隔离性、持久性的完整事务支持
- **高性能**: 支持高并发的事务处理，提供低延迟的响应
- **可扩展性**: 支持分布式事务，能够水平扩展
- **兼容性**: 支持 MySQL 兼容的事务隔离级别

### 1.2 核心特性

- **MVCC 模型**: 多版本并发控制，支持快照隔离
- **两阶段提交**: 分布式事务的两阶段提交协议
- **悲观锁和乐观锁**: 支持两种并发控制模式
- **死锁检测**: 自动检测和解决死锁
- **时间戳管理**: 基于 TSO 的时间戳服务

```rust
pub struct Key {

impl Key {
    // 编码 Key，添加时间戳和事务信息
    pub fn encode(&self, ts: u64) -> Vec<u8> {
        let mut encoded = Vec::with_capacity(self.raw_key.len() + 8);
        encoded.extend_from_slice(&self.raw_key);
        encoded.extend_from_slice(&ts.to_be_bytes());
        encoded
    }

    // 解码 Key
    pub fn decode(encoded: &[u8]) -> Result<(Key, u64)> {
        if encoded.len() < 8 {
            return Err(Error::InvalidKeyLength);
        }

        let raw_key_len = encoded.len() - 8;
        let raw_key = encoded[..raw_key_len].to_vec();
        let ts = u64::from_be_bytes(encoded[raw_key_len..].try_into().unwrap());

        Ok((Key::from_raw(raw_key), ts))
    }
}
```

```rust
impl VersionManager {
    // 添加新版本
    pub fn add_version(&self, key: &Key, version: MvccValue) -> Result<()> {
        // 1. 获取或创建版本链
        let mut chains = self.version_chains.write().unwrap();
        let chain = chains.entry(key.clone()).or_insert_with(VersionChain::new);

        // 2. 添加版本到链中
        chain.add_version(version)?;

        // 3. 检查是否需要压缩
        if chain.needs_compaction() {
            self.compact_version_chain(key, chain)?;
        }

        Ok(())
    }

    // 获取版本
    pub fn get_version(&self, key: &Key, ts: u64) -> Result<Option<MvccValue>> {
        let chains = self.version_chains.read().unwrap();

        if let Some(chain) = chains.get(key) {
            Ok(chain.get_version_at_ts(ts))
        } else {
            Ok(None)
        }
    }
}
```

#### 3.2.2 版本链实现

```rust
pub struct VersionChain {
    versions: Vec<MvccValue>,
    max_versions: usize,
}

impl VersionChain {
    pub fn new() -> Self {
        Self {
            versions: Vec::new(),
            max_versions: 1000, // 最大版本数量
        }
    }

    pub fn add_version(&mut self, version: MvccValue) -> Result<()> {
        // 1. 检查版本数量限制
        if self.versions.len() >= self.max_versions {
            self.compact_oldest_versions();
        }

        // 2. 按时间戳排序插入
        let pos = self.versions.partition_point(|v| v.start_ts > version.start_ts);
        self.versions.insert(pos, version);

        Ok(())
    }

    pub fn get_version_at_ts(&self, ts: u64) -> Option<MvccValue> {
        // 查找小于等于 ts 的最大版本
        for version in &self.versions {
            if version.start_ts <= ts && version.is_committed {
                return Some(version.clone());
            }
        }
        None
    }

    pub fn needs_compaction(&self) -> bool {
        self.versions.len() > self.max_versions / 2
    }

    pub fn compact_oldest_versions(&mut self) {
        // 保留最近的版本
        let keep_count = self.max_versions / 2;
        if self.versions.len() > keep_count {
            self.versions.truncate(keep_count);
        }
    }
}
```

```rust
        // 1. 检查事务状态

        if let Some(lock) = mvcc_value.lock {
            self.handle_lock_conflict(key, &lock)?;
        }

        // 4. 返回值
        Ok(mvcc_value.value)
    }
```

### 5.2 锁获取和释放

#### 5.2.1 锁获取

```rust
impl LockManager {
    pub fn acquire_lock(&self, request: LockRequest) -> Result<LockResult> {
        // 1. 检查死锁
        if self.deadlock_detector.would_cause_deadlock(&request)? {
            return Err(Error::DeadlockDetected);
        }

        // 2. 尝试获取锁
        let lock = self.try_acquire_lock(&request)?;

        // 3. 检查是否需要等待
        if lock.status == LockStatus::Waiting {
            // 加入等待队列
            self.wait_queue.add_waiter(&request, lock.clone())?;

            // 等待锁释放或超时
            self.wait_for_lock(&request)?;
        }

        Ok(LockResult {
            lock,
            acquired: true,
        })
    }

    fn try_acquire_lock(&self, request: &LockRequest) -> Result<Lock> {
        // 1. 检查锁兼容性
        if !self.is_lock_compatible(request)? {
            // 创建等待锁
            return Ok(self.create_waiting_lock(request));
        }

        // 2. 创建锁
        let lock = Lock {
            lock_type: request.lock_type,
            transaction_id: request.transaction_id,
            lock_time: Instant::now(),
            expire_time: Instant::now() + request.timeout,
            key_range: request.key_range.clone(),
            status: LockStatus::Active,
        };

        // 3. 存储锁
        self.lock_store.store_lock(&lock)?;

        Ok(lock)
    }
}
```

```rust
impl LockManager {
    pub fn release_lock(&self, transaction_id: TransactionId, key: &Key) -> Result<()> {
        // 1. 获取锁
        let lock = self.lock_store.get_lock(key)?;

        // 2. 验证锁所有权
        if lock.transaction_id != transaction_id {
            return Err(Error::LockNotOwned);
        }

        // 3. 释放锁
        self.lock_store.remove_lock(key)?;

        // 4. 通知等待者
        self.notify_waiters(key)?;

        Ok(())
    }

    fn notify_waiters(&self, key: &Key) -> Result<()> {
        // 1. 获取等待队列
        let waiters = self.wait_queue.get_waiters(key)?;

        // 2. 通知等待者
        for waiter in waiters {
            // 重新尝试获取锁
            if let Ok(lock_result) = self.acquire_lock(waiter.request.clone()) {
                // 通知锁获取成功
                waiter.notify_lock_acquired(lock_result.lock)?;
            }
        }

        Ok(())
    }
}
```

```rust
impl DeadlockDetector {
    pub fn would_cause_deadlock(&self, request: &LockRequest) -> Result<bool> {
        // 1. 构建当前等待图
        let graph = self.build_wait_graph()?;

        // 2. 添加新的等待关系
        let mut test_graph = graph.clone();
        test_graph.add_wait_edge(
            request.transaction_id,
            &request.key_range,
        )?;

        // 3. 检测环路
        Ok(self.detect_cycle(&test_graph)?)
    }

    fn detect_cycle(&self, graph: &WaitGraph) -> Result<bool> {
        // 使用深度优先搜索检测环路
        let mut visited = HashSet::new();
        let mut recursion_stack = HashSet::new();

        for node in graph.nodes.keys() {
            if !visited.contains(node) {
                if self.dfs_detect_cycle(node, graph, &mut visited, &mut recursion_stack)? {
                    return Ok(true);
                }
            }
        }

        Ok(false)
    }

    fn dfs_detect_cycle(
        &self,
        node: &TransactionId,
        graph: &WaitGraph,
        visited: &mut HashSet<TransactionId>,
        recursion_stack: &mut HashSet<TransactionId>,
    ) -> Result<bool> {
        visited.insert(*node);
        recursion_stack.insert(*node);

        // 访问邻居节点
        for neighbor in graph.get_neighbors(node) {
            if !visited.contains(neighbor) {
                if self.dfs_detect_cycle(neighbor, graph, visited, recursion_stack)? {
                    return Ok(true);
                }
            } else if recursion_stack.contains(neighbor) {
                return Ok(true);
            }
        }

        recursion_stack.remove(node);
        Ok(false)
    }
}
```

## 6. 两阶段提交

### 6.1 预提交阶段

```rust
impl Transaction {
    pub fn prewrite(&mut self) -> Result<()> {
        // 1. 获取提交时间戳
        self.commit_ts = Some(self.tso_client.get_timestamp()?);

        // 2. 执行预提交
        self.execute_prewrite()?;

        // 3. 更新事务状态
        self.status = TransactionStatus::Committing;

        Ok(())
    }

    fn execute_prewrite(&mut self) -> Result<()> {
        // 1. 获取所有写入键的锁
        for key in &self.write_set {
            self.acquire_write_lock(key)?;
        }

        // 2. 预写入 MVCC 版本
        for (key, value) in &self.write_buffer {
            self.prewrite_mvcc_version(key, value)?;
        }

        Ok(())
    }

    fn acquire_write_lock(&self, key: &Key) -> Result<()> {
        let lock_request = LockRequest {
            transaction_id: self.id,
            lock_type: LockType::Write,
            key_range: Some(KeyRange::single(key.clone())),
            timeout: Duration::from_secs(30),
        };

        let lock_result = self.lock_manager.acquire_lock(lock_request)?;
        self.locks.insert(key.clone(), lock_result.lock);

        Ok(())
    }

    fn prewrite_mvcc_version(&self, key: &Key, value: &Value) -> Result<()> {
        let mvcc_value = MvccValue {
            value: Some(value.clone()),
            start_ts: self.start_ts,
            commit_ts: self.commit_ts.unwrap(),
            transaction_id: self.id,
            flags: MvccFlags {
                is_locked: true,
                is_committed: false,
                is_rollback: false,
            },
        };

        self.mvcc_store.write_mvcc_version(key, &mvcc_value)?;
        Ok(())
    }
}
```

### 6.2 提交阶段

```rust
impl Transaction {
    pub fn commit(&mut self) -> Result<()> {
        // 1. 检查事务状态
        if self.status != TransactionStatus::Committing {
            return Err(Error::InvalidTransactionState);
        }

        // 2. 执行提交
        self.execute_commit()?;

        // 3. 释放锁
        self.release_all_locks()?;

        // 4. 更新事务状态
        self.status = TransactionStatus::Committed;

        // 5. 通知监听器
        self.notify_listeners(TransactionEvent::Commit);

        Ok(())
    }

    fn execute_commit(&self) -> Result<()> {
        // 1. 提交 MVCC 版本
        for key in &self.write_set {
            self.commit_mvcc_version(key)?;
        }

        // 2. 更新索引
        self.update_transaction_index()?;

        Ok(())
    }

    fn commit_mvcc_version(&self, key: &Key) -> Result<()> {
        // 1. 读取预写入的版本
        let mut mvcc_value = self.mvcc_store.get_mvcc_version(key, self.start_ts)?;

        // 2. 更新为已提交状态
        mvcc_value.flags.is_committed = true;
        mvcc_value.flags.is_locked = false;

        // 3. 写入提交后的版本
        self.mvcc_store.update_mvcc_version(key, &mvcc_value)?;

        Ok(())
    }
}
```

### 6.3 回滚处理

```rust
impl Transaction {
    pub fn rollback(&mut self) -> Result<()> {
        // 1. 检查事务状态
        if self.status == TransactionStatus::Committed {
            return Err(Error::CannotRollbackCommitted);
        }

        // 2. 执行回滚
        self.execute_rollback()?;

        // 3. 释放锁
        self.release_all_locks()?;

        // 4. 更新事务状态
        self.status = TransactionStatus::RolledBack;

        // 5. 通知监听器
        self.notify_listeners(TransactionEvent::Rollback);

        Ok(())
    }

    fn execute_rollback(&self) -> Result<()> {
        // 1. 回滚 MVCC 版本
        for key in &self.write_set {
            self.rollback_mvcc_version(key)?;
        }

        // 2. 清理写入缓冲区
        self.write_buffer.clear();

        Ok(())
    }

    fn rollback_mvcc_version(&self, key: &Key) -> Result<()> {
        // 1. 读取预写入的版本
        if let Some(mut mvcc_value) = self.mvcc_store.get_mvcc_version(key, self.start_ts)? {
            // 2. 标记为回滚
            mvcc_value.flags.is_rollback = true;
            mvcc_value.flags.is_locked = false;

            // 3. 更新版本
            self.mvcc_store.update_mvcc_version(key, &mvcc_value)?;
        }

        Ok(())
    }
}
```

```rust
        }

        // 2. 从 TSO 服务获取
        self.get_timestamp_from_tso()
    }

    fn try_get_from_cache(&self) -> Option<u64> {
        let mut cache = self.timestamp_cache.lock().unwrap();

        if cache.current <= cache.end {
            let ts = cache.current;
            cache.current += 1;
            Some(ts)
        } else {
            None
        }
    }

    fn get_timestamp_from_tso(&self) -> Result<u64> {
        // 1. 从 PD 批量获取时间戳
        let (start, end) = self.pd_client.get_timestamp_batch(1000)?;

        // 2. 更新缓存
        {
            let mut cache = self.timestamp_cache.lock().unwrap();
            cache.start = start;
            cache.end = end;
            cache.current = start;
        }

        // 3. 返回第一个时间戳
        Ok(start)
    }
}
```

```rust
impl RecoveryManager {
    pub async fn recover_transaction(&self, transaction_id: TransactionId) -> Result<()> {
        // 1. 恢复事务状态
        let transaction = self.recover_transaction_state(transaction_id)?;

        // 2. 检查事务状态
        match transaction.status {
            TransactionStatus::Committing => {
                // 重新提交
                self.recover_committing_transaction(&transaction).await?;
            }
            TransactionStatus::RollingBack => {
                // 重新回滚
                self.recover_rollingback_transaction(&transaction).await?;
            }
            TransactionStatus::Active => {
                // 超时回滚
                self.recover_active_transaction(&transaction).await?;
            }
            _ => {
                // 其他状态无需恢复
            }
        }

        Ok(())
    }
```
