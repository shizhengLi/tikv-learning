# TiKV 协处理器模块详解

## 1. 模块概述

TiKV 的协处理器（Coprocessor）模块是一个强大的计算下推框架，允许将部分计算逻辑下推到存储层执行。这种设计显著提高了查询性能，减少了网络传输开销，并支持了分布式计算能力。

### 1.1 设计目标

- **计算下推**: 将计算任务下推到数据存储位置执行
- **性能优化**: 减少网络传输和数据移动
- **可扩展性**: 支持自定义计算逻辑的插件化扩展
- **分布式计算**: 支持跨节点的并行计算

### 1.2 核心特性

- **DAG 执行引擎**: 基于有向无环图的执行引擎
- **表达式计算**: 支持复杂表达式的计算和优化
- **聚合函数**: 内置丰富的聚合函数支持
- **索引扫描**: 高效的索引扫描和范围查询
- **结果缓存**: 智能的结果缓存机制
- **资源控制**: 完善的资源控制和限流机制

```rust
impl DAGBuilder {
    pub fn build_graph(&self, request: &DAGRequest) -> Result<ExecutionGraph> {
        // 1. 解析请求
        let parsed_request = self.parse_request(request)?;

        // 2. 创建执行节点
        let nodes = self.create_execution_nodes(&parsed_request)?;

        // 3. 建立节点连接
        let edges = self.create_edges(&nodes, &parsed_request)?;

        // 4. 优化执行图
        let optimized_graph = self.optimize_graph(ExecutionGraph { nodes, edges })?;

        Ok(optimized_graph)
    }

    fn create_execution_nodes(&self, request: &ParsedRequest) -> Result<HashMap<NodeId, ExecutionNode>> {
        let mut nodes = HashMap::new();
        let mut node_id = 0;

        // 创建数据源节点
        for source in &request.data_sources {
            let node = ExecutionNode {
                id: NodeId(node_id),
                node_type: NodeType::DataSource,
                executor: Arc::new(DataSourceExecutor::new(source.clone())),
                inputs: Vec::new(),
                outputs: Vec::new(),
                resource_requirements: ResourceRequirements::default(),
            };
            nodes.insert(NodeId(node_id), node);
            node_id += 1;
        }

        // 创建处理节点
        for operation in &request.operations {
            let node = self.create_operation_node(operation, node_id)?;
            nodes.insert(NodeId(node_id), node);
            node_id += 1;
        }

        Ok(nodes)
    }
}
```

```rust
    pub fn optimize_plan(&self, graph: ExecutionGraph) -> Result<ExecutionPlan> {
        // 1. 拓扑排序
        let execution_order = self.topological_sort(&graph)?;

        // 2. 并行化分析
        let parallel_strategy = self.analyze_parallelism(&graph, &execution_order)?;

        // 3. 资源分配
        let resource_allocation = self.allocate_resources(&graph, &parallel_strategy)?;

        // 4. 缓存策略制定

        Ok(ExecutionPlan {

            execution_order,
            resource_allocation,
            parallel_strategy,
            cache_strategy,
        })
    }

    fn topological_sort(&self, graph: &ExecutionGraph) -> Result<Vec<NodeId>> {
        let mut visited = HashSet::new();
        let mut order = Vec::new();
        let mut stack = Vec::new();

        // 从源节点开始遍历
        for source_id in &graph.sources {
            if !visited.contains(source_id) {
                stack.push(*source_id);
                while let Some(node_id) = stack.pop() {
                    if !visited.contains(&node_id) {
                        visited.insert(node_id);
                        order.push(node_id);

                        // 添加未访问的邻居节点
                        for neighbor in self.get_neighbors(&graph, node_id) {
                            if !visited.contains(&neighbor) {
                                stack.push(neighbor);
                            }
                        }
                    }
                }
            }
        }

        Ok(order)
    }
}
```

```rust
impl DAGExecutor {
    pub async fn execute(&self, request: DAGRequest) -> Result<QueryResult> {
        // 1. 构建执行图
        let graph = self.build_graph(&request)?;

        // 2. 生成执行计划
        let plan = self.plan_execution(graph)?;

        // 3. 分配资源
        let resources = self.allocate_resources(&plan)?;

        // 4. 执行查询
        let result = self.execute_plan(plan, resources).await?;

        // 5. 更新统计信息
        self.update_statistics(&result);

        Ok(result)
    }

    async fn execute_plan(&self, plan: ExecutionPlan, resources: ResourceAllocation) -> Result<QueryResult> {
        // 1. 创建执行上下文
        let context = ExecutionContext::new(plan.clone(), resources);

        // 2. 并行执行
        let handles = plan.execution_order
            .into_par_iter()
            .map(|node_id| {
                let context = context.clone();
                tokio::spawn(async move {
                    self.execute_node(node_id, &context).await
                })
            })
            .collect::<Vec<_>>();

        // 3. 等待所有任务完成
        let results: Result<Vec<_>> = futures::future::join_all(handles)
            .await
            .into_iter()
            .map(|r| r.unwrap())
            .collect();

        // 4. 合并结果
        self.merge_results(results?)
    }

    async fn execute_node(&self, node_id: NodeId, context: &ExecutionContext) -> Result<NodeResult> {
        // 1. 获取节点
        let node = context.plan.graph.get_node(node_id)?;

        // 2. 准备输入数据
        let input_data = self.prepare_input_data(node, context).await?;

        // 3. 执行节点
        let result = node.executor.execute(context)?;

        // 4. 传递给下游节点
        self.pass_to_downstream(node_id, result, context).await?;

        Ok(NodeResult::Success)
    }
}
```

```rust
impl ExpressionParser {
    pub fn parse(&self, expr_str: &str) -> Result<Expression> {
        // 1. 词法分析
        let tokens = self.tokenize(expr_str)?;

        // 2. 语法分析
        let ast = self.parse_tokens(&tokens)?;

        // 3. 语义分析
        let expression = self.analyze_semantics(ast)?;

        Ok(expression)
    }

    fn tokenize(&self, expr_str: &str) -> Result<Vec<Token>> {
        let mut tokens = Vec::new();
        let mut chars: Vec<char> = expr_str.chars().collect();
        let mut pos = 0;

        while pos < chars.len() {
            let ch = chars[pos];

            match ch {
                // 跳过空白字符
                ' ' | '\t' | '\n' => {
                    pos += 1;
                }

                // 数字
                '0'..='9' => {
                    let (token, new_pos) = self.parse_number(&chars, pos);
                    tokens.push(token);
                    pos = new_pos;
                }

                // 标识符
                'a'..='z' | 'A'..='Z' | '_' => {
                    let (token, new_pos) = self.parse_identifier(&chars, pos);
                    tokens.push(token);
                    pos = new_pos;
                }

                // 操作符
                '+' | '-' | '*' | '/' | '=' | '!' | '<' | '>' | '&' | '|' => {
                    let (token, new_pos) = self.parse_operator(&chars, pos);
                    tokens.push(token);
                    pos = new_pos;
                }

                // 括号和逗号
                '(' | ')' | ',' => {
                    tokens.push(Token::Punctuation(ch));
                    pos += 1;
                }

                // 字符串
                '"' | '\'' => {
                    let (token, new_pos) = self.parse_string(&chars, pos);
                    tokens.push(token);
                    pos = new_pos;
                }

                _ => {
                    return Err(Error::InvalidCharacter(ch));
                }
            }
        }

        Ok(tokens)
    }
}
```

```rust
impl ExpressionCompiler {
    pub fn compile(&self, expression: &Expression) -> Result<CompiledExpression> {
        match expression {
            Expression::Constant(value) => self.compile_constant(value),
            Expression::ColumnRef(col_ref) => self.compile_column_ref(col_ref),
            Expression::FunctionCall(call) => self.compile_function_call(call),
            Expression::Operator(op_expr) => self.compile_operator(op_expr),
            Expression::Aggregate(agg_expr) => self.compile_aggregate(agg_expr),
            Expression::Conditional(cond_expr) => self.compile_conditional(cond_expr),
        }
    }

    fn compile_function_call(&self, call: &FunctionCall) -> Result<CompiledExpression> {
        // 1. 编译参数
        let compiled_args: Vec<CompiledExpression> = call.arguments
            .iter()
            .map(|arg| self.compile(arg))
            .collect::<Result<Vec<_>>>()?;

        // 2. 获取函数实现
        let function_impl = self.function_registry.get_function(&call.function_name)?;

        // 3. 创建编译后的表达式
        Ok(CompiledExpression::FunctionCall {
            function: function_impl,
            arguments: compiled_args,
            return_type: call.return_type.clone(),
        })
    }
}
```

```rust
impl ExpressionEngine {
    pub fn evaluate(&self, expression: &CompiledExpression, context: &ExecutionContext) -> Result<Value> {
        match expression {
            CompiledExpression::Constant(value) => Ok(value.clone()),
            CompiledExpression::ColumnRef(col_ref) => self.evaluate_column_ref(col_ref, context),
            CompiledExpression::FunctionCall { function, arguments, .. } => {
                self.evaluate_function_call(function, arguments, context)
            }
            CompiledExpression::Operator { op, left, right } => {
                self.evaluate_operator(op, left, right, context)
            }
            CompiledExpression::Aggregate { agg_func, args } => {
                self.evaluate_aggregate(agg_func, args, context)
            }
            CompiledExpression::Conditional { condition, then_expr, else_expr } => {
                self.evaluate_conditional(condition, then_expr, else_expr, context)
            }
        }
    }

    fn evaluate_function_call(
        &self,
        function: &Arc<dyn Function>,
        arguments: &[CompiledExpression],
        context: &ExecutionContext,
    ) -> Result<Value> {
        // 1. 评估参数
        let evaluated_args: Vec<Value> = arguments
            .iter()
            .map(|arg| self.evaluate(arg, context))
            .collect::<Result<Vec<_>>>()?;

        // 2. 执行函数
        function.execute(&evaluated_args, context)
    }
}
```

```rust
impl AggregateFunction for SumFunction {
    fn name(&self) -> &str {
        "SUM"
    }

        Box::new(SumState::new(self.data_type.clone()))
    }

    fn update_state(&self, state: &mut dyn AggregateState, value: &Value) -> Result<()> {

            .ok_or(Error::InvalidStateType)?;

        Ok(())

    fn merge_states(&self, target: &mut dyn AggregateState, source: &dyn AggregateState) -> Result<()> {

        let source_state = source.as_any().downcast::<SumState>()
            .ok_or(Error::InvalidStateType)?;

        target_state.merge(source_state)?;
        Ok(())
    }

    fn get_result(&self, state: &dyn AggregateState) -> Result<Value> {
        let sum_state = state.as_any().downcast_ref::<SumState>()
            .ok_or(Error::InvalidStateType)?;
        sum_state.get_result()
    }

    fn return_type(&self) -> DataType {
        self.data_type.clone()
    }
}

pub struct SumState {
    sum: Value,
    count: u64,
    data_type: DataType,
}

impl SumState {
    pub fn new(data_type: DataType) -> Self {
        Self {
            sum: Value::zero(&data_type),
            count: 0,
            data_type,
        }
    }

    pub fn add_value(&mut self, value: &Value) -> Result<()> {
        self.sum = self.sum.add(value)?;
        self.count += 1;
        Ok(())
    }

    pub fn merge(&mut self, other: &SumState) -> Result<()> {
        self.sum = self.sum.add(&other.sum)?;
        self.count += other.count;
        Ok(())
    }

    pub fn get_result(&self) -> Result<Value> {
        if self.count == 0 {
            Ok(Value::null())
        } else {
            Ok(self.sum.clone())
        }
    }
}
```

## 6. 请求调度

### 6.1 请求队列

```rust
pub struct RequestQueue {
    // 待处理请求
    pending_requests: Arc<Mutex<VecDeque<DAGRequest>>>,

    // 优先级队列
    priority_queue: Arc<Mutex<PriorityQueue<DAGRequest>>>,

    // 资源限制
    resource_limits: ResourceLimits,

    // 通知机制
    notification: Arc<Notification>,
}

impl RequestQueue {
    pub fn enqueue(&self, request: DAGRequest) -> Result<()> {
        // 1. 检查资源限制
        if !self.check_resource_limits(&request)? {
            return Err(Error::ResourceLimitExceeded);
        }

        // 2. 根据优先级排队
        match request.priority {
            Priority::High => {
                let mut queue = self.priority_queue.lock().unwrap();
                queue.push(request);
            }
            Priority::Normal => {
                let mut queue = self.pending_requests.lock().unwrap();
                queue.push_back(request);
            }
            Priority::Low => {
                let mut queue = self.pending_requests.lock().unwrap();
                queue.push_front(request);
            }
        }

        // 3. 通知调度器
        self.notification.notify();

        Ok(())
    }

    pub fn dequeue(&self) -> Option<DAGRequest> {
        // 1. 优先处理高优先级请求
        if let Some(request) = self.priority_queue.lock().unwrap().pop() {
            return Some(request);
        }

        // 2. 处理普通请求
```

```rust
        // 简单估算：基于操作数量和数据量
        let op_count = request.operations.len() as u64;
        let data_size = request.estimated_data_size();
        Duration::from_millis(op_count * 10 + data_size / 1024)
    }

        // 检查内存和 CPU 资源
        request.required_memory() < self.available_memory() &&
        request.required_cpu() < self.available_cpu()
    }
```

```rust
            return None;
        }

        // 2. 尝试从缓存获取
        if let Some(result) = self.cache_store.get(cache_key) {
            // 3. 检查结果是否过期
            if self.is_result_fresh(&result) {

            }
        }

        self.cache_stats.miss();
        None
    }

        // 1. 检查是否应该缓存
        if !self.should_cache(&result) {
            return Ok(());
        }

        // 2. 存储到缓存
        self.cache_store.put(cache_key, result.clone())?;

        // 3. 更新统计信息
        self.cache_stats.put();

        Ok(())
    }
}
```

```rust
    pub fn load_plugin(&self, plugin_path: &Path) -> Result<()> {
        // 1. 加载插件库
        let plugin = self.loader.load(plugin_path)?;

        // 2. 初始化插件
        let context = PluginContext::new();
        plugin.initialize(&context)?;

        // 3. 注册插件
        {
            let mut plugins = self.plugins.write().unwrap();
            plugins.insert(plugin.name().to_string(), plugin);
        }

        Ok(())
    }

    pub fn unload_plugin(&self, plugin_name: &str) -> Result<()> {
        // 1. 获取插件
        let plugin = {
            let mut plugins = self.plugins.write().unwrap();
            plugins.remove(plugin_name)
                .ok_or(Error::PluginNotFound(plugin_name.to_string()))?
        };

        // 2. 清理插件
        plugin.cleanup()?;

        Ok(())
    }

    pub fn execute_plugin(&self, plugin_name: &str, request: PluginRequest) -> Result<PluginResponse> {
        // 1. 获取插件
        let plugins = self.plugins.read().unwrap();
        let plugin = plugins.get(plugin_name)
            .ok_or(Error::PluginNotFound(plugin_name.to_string()))?;

        // 2. 执行插件
        plugin.handle_request(&request)
    }
}
```

```rust
    thread_pool: Arc<ThreadPool>,

impl ParallelExecutor {
    pub async fn execute_parallel(&self, tasks: Vec<ExecutionTask>) -> Result<Vec<TaskResult>> {
        // 1. 创建并行任务
        let futures: Vec<_> = tasks
            .into_iter()
            .map(|task| {
                let thread_pool = self.thread_pool.clone();
                async move {
                    thread_pool.execute(task).await
                }
            })
            .collect();

        // 2. 并行执行
        let results: Result<Vec<_>> = futures::future::try_join_all(futures).await;

        // 3. 收集结果
        self.result_collector.collect_results(results?)
    }
}
```
